+	}
+	got, err := controllerEdgeCandidates(conn, mappings)
+	if err != nil {
+		t.Fatal(err)
+	}
+
+	wantPredicted := map[string]bool{
+		"203.0.113.10:51004": false,
+		"203.0.113.10:51005": false,
+		"203.0.113.10:51006": false,
+	}
+	for _, candidate := range got {
+		if _, ok := wantPredicted[candidate.Address]; ok {
+			wantPredicted[candidate.Address] = true
+			if candidate.Priority >= 1200 {
+				t.Fatalf("predicted candidate has exact-mapping priority: %+v", candidate)
+			}
+		}
+	}
+	for address, found := range wantPredicted {
+		if !found {
+			t.Fatalf("predicted candidate missing: %s; got=%+v", address, got)
+		}
+	}

+func TestControllerEdgeCandidatesDoesNotPredictIrregularNAT(t *testing.T) {
+	conn, err := net.ListenUDP("udp4", &net.UDPAddr{IP: net.IPv4zero, Port: 0})
+	if err != nil {
+		t.Fatal(err)
+	}
+	defer conn.Close()
+
+	mappings := []edgeproto.Mapping{
+		{Server: "stun-a:3478", Address: "203.0.113.10:51001"},
+		{Server: "stun-b:3478", Address: "203.0.113.10:51004"},
+		{Server: "stun-c:3478", Address: "203.0.113.10:51006"},
+	}
+	got, err := controllerEdgeCandidates(conn, mappings)
+	if err != nil {
+		t.Fatal(err)
+	}
+
+	publicCount := 0
+	for _, candidate := range got {
+		host, _, splitErr := net.SplitHostPort(candidate.Address)
+		if splitErr == nil && host == "203.0.113.10" {
+			publicCount++
+		}
+	}
+	if publicCount != len(mappings) {
+		t.Fatalf("irregular NAT produced guessed public candidates: count=%d want=%d got=%+v", publicCount, len(mappings), got)
+	}
